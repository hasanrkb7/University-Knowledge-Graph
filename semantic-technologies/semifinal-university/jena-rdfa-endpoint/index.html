💡 Example Queries

List all students with details

PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
SELECT ?student ?name ?email ?gpa
WHERE {
    ?student rdf:type :Student .
    ?student :hasName ?name .
    ?student :hasEmail ?email .
    ?student :hasGPA ?gpa .
}

Find courses taught by instructors

SELECT ?course ?instructor
WHERE {
    ?course :teaches ?instructor .
    ?instructor rdf:type :Instructor .
}

Calculate average GPA

SELECT (AVG(?gpa) AS ?averageGPA)
WHERE {
    ?student :hasGPA ?gpa .
}
